cd "$PBS_O_WORKDIR"

# Force underlying C/C++ math libraries to use all 24 requested CPUs
export OMP_NUM_THREADS=24
export MKL_NUM_THREADS=24
export OPENBLAS_NUM_THREADS=24

# Initialize Conda
source ~/miniforge3/etc/profile.d/conda.sh
conda activate pnm

# Run Python inside a Virtual Framebuffer to prevent Animation/Plotting crashes
xvfb-run -a python single_sim_run.py > verbose_log.txt 2>&1
